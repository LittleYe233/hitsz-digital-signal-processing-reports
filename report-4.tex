\textbf{问题 1}\quad 图 4-1 中, 以 $x(n)$ 为系统输入, $x_0(n)$ 为系统输出, 写出系统的差分方程与系统函数 $H(z)$.

\begin{block}
\begin{gather*}
    x_0(n)=x(n)-a_1x_0(n-1)-a_2x_0(n-2). \\
    H(z)=\frac{x_0(n)}{x(n)}=\frac{1}{1+a_1z^{-1}+a_2z^{-2}}.
\end{gather*}
\end{block}

\begin{block}
\begin{gather*}
    y(n)=x(n)+b_0x_0(n)+b_1x_0(n-1)+b_2x_0(n-2)-a_1y(n-1)-a_2y(n-2). \\
    H(z)=\frac{b_0+b_1z^{-1}+b_2z^{-2}}{1+a_1z^{-1}+a_2z^{-2}}.
\end{gather*}
\end{block}
